local function getRoleRaw(plr)
    if not plr or not plr.Character then return nil end
    local d = roundData(plr)
    if d and d.Role then
        local r = d.Role
        if r == "Murderer" then return "Murderer" end
        if r == "Sheriff" or r == "Hero" then return "Sheriff" end
        if r == "Innocent" then return "Innocent" end
    end
    local ch = plr.Character
    local bp = plr:FindFirstChild("Backpack")
    local function hasTag(obj, tag)
        if not obj then return false end
        return CollectionService:HasTag(obj, tag)
    end
    if hasTag(ch, "Weapon_Gun") or hasTag(bp, "Weapon_Gun") then return "Sheriff" end
    if hasTag(ch, "Weapon_Knife") or hasTag(bp, "Weapon_Knife") then return "Murderer" end
    local hasKnife, hasGun = false, false
    local function checkTool(tool)
        if not tool or not tool:IsA("Tool") then return end
        local name = tool.Name:lower()
        if name:find("knife") or name:find("murderer") then hasKnife = true end
        if name:find("gun") or name:find("revolver") or name:find("sheriff") then hasGun = true end
    end
    if bp then
        for _, tool in ipairs(bp:GetChildren()) do checkTool(tool) end
    end
    checkTool(ch:FindFirstChildOfClass("Tool"))
    if hasKnife then return "Murderer" end
    if hasGun then return "Sheriff" end
    return "Innocent"
end

local function getRole(plr)
    if not plr then return nil end
    local c = roleCache[plr]
    local now = tick()
    if c and now - c.time < ROLE_CACHE_TIME then return c.role end
    local role = getRoleRaw(plr)
    roleCache[plr] = {role = role, time = now}
    return role
end

local function freshCoinsRaw()
    local out = {}
    local tagged = CollectionService:GetTagged("ServerCoinPart")
    if #tagged > 0 then
        for _, d in ipairs(tagged) do
            if d:IsA("BasePart") and d.Parent and not coinTaken(d) then
                out[#out + 1] = d
            end
        end
        return out
    end
    local map = CollectionService:GetTagged("CurrentMap")[1]
    local container = (map and map:FindFirstChild("CoinContainer")) 
                   or workspace:FindFirstChild("CoinContainer", true)
    if container then
        for _, d in ipairs(container:GetChildren()) do
            if d:IsA("BasePart") and (d:GetAttribute("CoinID") ~= nil or d.Name == "Coin_Server") and not coinTaken(d) then
                out[#out + 1] = d
            end
        end
    end
    return out
end

local function freshCoins()
    local now = tick()
    if coinCache.list and now - coinCache.time < COIN_CACHE_TIME then
        local valid = {}
        for _, c in ipairs(coinCache.list) do
            if c.Parent and not coinTaken(c) then valid[#valid + 1] = c end
        end
        if #valid > 0 then return valid end
    end
    local list = freshCoinsRaw()
    coinCache = {list = list, time = now}
    return list
end

local function findNearestCoin(root)
    local coins = freshCoins()
    local nearest, nearestDist = nil, math.huge
    for _, c in ipairs(coins) do
        if c.Parent then
            local d = (c.Position - root.Position).Magnitude
            if d < nearestDist then nearestDist = d; nearest = c end
        end
    end
    return nearest, #coins
end

local AIM_CONFIG = {
    VelocityAlpha = 0.70,
    AccelerationAlpha = 0.25,
    MinLead = 0.03,
    MaxLead = 0.30,
    MaxVelocity = 150,
    MaxAcceleration = 100,
    GroundLeadMultiplier = 1.0,
    AirLeadMultiplier = 0.75,
    ReactionTime = 0.02,
    UpdateEveryNFrames = 2,
}

local aimStates = {}
local aimFrameCounter = 0
local raycastCache = {lastAt = 0}

local function clampVecMag(v, maxMag)
    local mag = v.Magnitude
    if mag > maxMag and mag > 0 then
        return v.Unit * maxMag
    end
    return v
end

local function updateAimState(plr, root, dt)
    if not plr or not root or dt <= 0 then return nil end
    local uid = plr.UserId
    local s = aimStates[uid]
    local pos = root.Position
    
    if not s then
        s = {
            Position = pos,
            Velocity = Vector3.zero,
            Acceleration = Vector3.zero,
            LastUpdate = tick(),
        }
        aimStates[uid] = s
        return s
    end
    
    local rawVel = (pos - s.Position) / dt
    rawVel = clampVecMag(rawVel, AIM_CONFIG.MaxVelocity)
    
    local smoothVel
    if s.Velocity.Magnitude < 0.01 then
        smoothVel = rawVel
    else
        smoothVel = s.Velocity:Lerp(rawVel, AIM_CONFIG.VelocityAlpha)
    end
    
    local rawAccel = (smoothVel - s.Velocity) / dt
    rawAccel = clampVecMag(rawAccel, AIM_CONFIG.MaxAcceleration)
    
    local smoothAccel = s.Acceleration:Lerp(rawAccel, AIM_CONFIG.AccelerationAlpha)
    
    s.Position = pos
    s.Velocity = smoothVel
    s.Acceleration = smoothAccel
    s.LastUpdate = tick()
    
    return s
end

local function calculateLeadTime(pingMs)
    local ping = tonumber(pingMs) or 100
    local networkTime = (ping / 1000) * 0.5
    local lead = networkTime + AIM_CONFIG.ReactionTime
    lead = math.clamp(lead, AIM_CONFIG.MinLead, AIM_CONFIG.MaxLead)
    return lead
end

local function predictPosition(pos, vel, accel, t)
    return pos + vel * t + accel * (0.5 * t * t)
end

local function isFlingVictim(root)
    if not root then return false end
    return root.AssemblyLinearVelocity.Magnitude > AIM_CONFIG.MaxVelocity
end

task.spawn(function()
    while true do
        local dt = RunService.Heartbeat:Wait()
        aimFrameCounter = aimFrameCounter + 1
        
        if aimFrameCounter % AIM_CONFIG.UpdateEveryNFrames ~= 0 then
            continue
        end
        
        local updateDt = dt * AIM_CONFIG.UpdateEveryNFrames
        
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= player and p.Character then
                local root = p.Character:FindFirstChild("HumanoidRootPart")
                if root then
                    pcall(updateAimState, p, root, updateDt)
                end
            end
        end
    end
end)

local function calculateAim(targetChar, targetRoot, targetHum)
    if not targetChar or not targetRoot or not targetHum then
        return nil, "INVALID"
    end
    if targetHum.Health <= 0 then
        return nil, "DEAD"
    end
    
    local plr = Players:GetPlayerFromCharacter(targetChar)
    if not plr then
        return targetRoot.Position, "FALLBACK"
    end
    
    local s = aimStates[plr.UserId]
    local now = tick()
    local dt = 1/60
    if s then
        dt = math.clamp(now - s.LastUpdate, 1/120, 0.1)
    end
    
    s = updateAimState(plr, targetRoot, dt)
    if not s then
        return targetRoot.Position, "NO_STATE"
    end
    
    if isFlingVictim(targetRoot) then
        return s.Position, "FLING_VICTIM"
    end
    
    local ping = getPing() * 1000
    local lead = calculateLeadTime(ping)
    
    local hstate = targetHum:GetState()
    local multiplier = AIM_CONFIG.GroundLeadMultiplier
    if hstate == Enum.HumanoidStateType.Jumping 
       or hstate == Enum.HumanoidStateType.Freefall then
        multiplier = AIM_CONFIG.AirLeadMultiplier
    end
    lead = lead * multiplier
    
    local predicted = predictPosition(
        targetRoot.Position,
        s.Velocity,
        s.Acceleration,
        lead
    )
    
    local distanceFromActual = (predicted - targetRoot.Position).Magnitude
    if distanceFromActual > 30 then
        predicted = targetRoot.Position + s.Velocity * lead
        return predicted, "SAFE_FALLBACK"
    end
    
    return predicted, "PREDICTED"
end

local function wallbangShoot(targetRoot)
    if not targetRoot or not targetRoot.Parent then return false end
    local char, myRoot = getChar()
    if not char or not myRoot then return false end
    local gun = getGun()
    if not gun or not gun:FindFirstChild("Shoot") then
        cachedGun = nil
        return false
    end
    local targetChar = targetRoot.Parent
    local targetHum = targetChar:FindFirstChild("Humanoid")
    if not targetHum or targetHum.Health <= 0 then return false end
    local aimPos, mode = calculateAim(targetChar, targetRoot, targetHum)
    if not aimPos then return false end
    local fakeOrigin = aimPos + Vector3.new(0, 3, 0)
    local fakeCFrame = CFrame.new(fakeOrigin, aimPos)
    local ok = pcall(function() gun.Shoot:FireServer(fakeCFrame, CFrame.new(aimPos)) end)
    if ok then
        lastShotFiredAt = tick()
    end
    return ok
end

local function trackPlayer(p)
    trackPlayerConnections[p] = trackPlayerConnections[p] or {}
    local conns = trackPlayerConnections[p]
    local charConn, diedConn, ancestryConn
    local function onChar(char)
        if diedConn then pcall(function() if diedConn.Connected then diedConn:Disconnect() end end) diedConn = nil end
        cachedGun = nil
        deadPlayers[p.UserId] = nil
        aimStates[p.UserId] = nil
        local hum = char:WaitForChild("Humanoid", 10)
        if hum then
            diedConn = hum.Died:Connect(function() deadPlayers[p.UserId] = true end)
            table.insert(conns, diedConn)
        end
    end
    charConn = p.CharacterAdded:Connect(onChar)
    table.insert(conns, charConn)
    if p.Character then pcall(onChar, p.Character) end
    ancestryConn = p.AncestryChanged:Connect(function()
        if not p:IsDescendantOf(game) then
            for _, c in ipairs(conns) do pcall(function() if c.Connected then c:Disconnect() end end) end
            trackPlayerConnections[p] = nil
            deadPlayers[p.UserId] = nil
            targetHistory[p.UserId] = nil
            roleCache[p] = nil
            aimStates[p.UserId] = nil
        end
    end)
    table.insert(conns, ancestryConn)
end

local function computeESPColor(p)

    if State.ESPFull then
        if role == "Murderer" then return Color3.fromRGB(255, 0, 0) end
        if role == "Sheriff" then return Color3.fromRGB(0, 100, 255) end
        if role == "Innocent" then return Color3.fromRGB(0, 255, 0) end
        return nil
    end
    if State.ESPAllGreen then
        return Color3.fromRGB(0, 255, 0)
    end
    if State.ESPMurderer and role == "Murderer" then
        return Color3.fromRGB(255, 0, 0)
    end
    if State.ESPSheriff and role == "Sheriff" then
        return Color3.fromRGB(0, 100, 255)
    end
    return nil
end

local function updateESP()
    for _, p in ipairs(Players:GetPlayers()) do
        if p == player then continue end
        local char = p.Character
        if not char then
            if highlights[p] then pcall(function() highlights[p]:Destroy() end) highlights[p] = nil end
            continue
        end
        local color = computeESPColor(p)
        if not color then
            if highlights[p] then pcall(function() highlights[p]:Destroy() end) highlights[p] = nil end
            continue
        end
        local hl = highlights[p]
        if hl and hl.Parent and hl.Adornee == char then
            if hl.FillColor ~= color then hl.FillColor = color; hl.OutlineColor = color end
        else
            if hl then pcall(function() hl:Destroy() end) end
            local newHl = Instance.new("Highlight")
            newHl.Name = "cCraftESP"
            newHl.FillColor = color; newHl.OutlineColor = color
            newHl.FillTransparency = 0.4; newHl.OutlineTransparency = 0
            newHl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            newHl.Adornee = char; newHl.Parent = char
            highlights[p] = newHl
        end
    end
end

local function cleanupESP()
    for p, hl in pairs(highlights) do if hl then pcall(function() hl:Destroy() end) end end
    highlights = {}
end

    RunId.ESP = RunId.ESP + 1

    end
end

local antiFlingEnabled = false

local function saveOriginalCanCollide(character)
    if not originalCanCollide[character] then originalCanCollide[character] = {} end
    for _, part in ipairs(character:GetChildren()) do
        if part:IsA("BasePart") then
            if originalCanCollide[character][part] == nil then
                originalCanCollide[character][part] = part.CanCollide
            end
        end
    end
end

local function disableAntiFling_Part(part)
    if antiFlingEnabled and part:IsA("BasePart") then part.CanCollide = false end
end

local function setupAntiFlingCharacter(character)
    saveOriginalCanCollide(character)
    local childAddedConn, steppedConn
    for _, part in ipairs(character:GetChildren()) do disableAntiFling_Part(part) end
    childAddedConn = character.ChildAdded:Connect(function(part)
        if part:IsA("BasePart") then
            if not originalCanCollide[character] then originalCanCollide[character] = {} end
            if originalCanCollide[character][part] == nil then
                originalCanCollide[character][part] = part.CanCollide
            end
        end
        disableAntiFling_Part(part)
    end)
    steppedConn = RunService.Stepped:Connect(function()
        if antiFlingEnabled and character:IsDescendantOf(workspace) then
            for _, part in ipairs(character:GetChildren()) do
                if part:IsA("BasePart") and part.CanCollide then part.CanCollide = false end
            end
        end
    end)
    character.Destroying:Connect(function()
        pcall(function() if childAddedConn.Connected then childAddedConn:Disconnect() end end)
        pcall(function() if steppedConn.Connected then steppedConn:Disconnect() end end)
    end)
end

local function trackAntiFlingPlayer(p)
    if p == player then return end
    local conn = p.CharacterAdded:Connect(setupAntiFlingCharacter)
    if p.Character then setupAntiFlingCharacter(p.Character) end
    antiFlingConnections[p] = conn
end

local function untrackAntiFlingPlayer(p)
    if antiFlingConnections[p] then
        pcall(function() if antiFlingConnections[p].Connected then antiFlingConnections[p]:Disconnect() end end)
        antiFlingConnections[p] = nil
    end
end

local function enableAntiFling()
    if antiFlingEnabled then return end
    antiFlingEnabled = true
    for _, p in ipairs(Players:GetPlayers()) do trackAntiFlingPlayer(p) end
    Connections.AntiFling_PlayerAdded = Players.PlayerAdded:Connect(trackAntiFlingPlayer)
    Connections.AntiFling_PlayerRemoving = Players.PlayerRemoving:Connect(untrackAntiFlingPlayer)
end

local function disableAntiFling()
    antiFlingEnabled = false
    if Connections.AntiFling_PlayerAdded then
        pcall(function() if Connections.AntiFling_PlayerAdded.Connected then Connections.AntiFling_PlayerAdded:Disconnect() end end)
        Connections.AntiFling_PlayerAdded = nil
    end
    if Connections.AntiFling_PlayerRemoving then
        pcall(function() if Connections.AntiFling_PlayerRemoving.Connected then Connections.AntiFling_PlayerRemoving:Disconnect() end end)
        Connections.AntiFling_PlayerRemoving = nil
    end
    for p, conn in pairs(antiFlingConnections) do
        pcall(function() if conn.Connected then conn:Disconnect() end end)
    end
    antiFlingConnections = {}
    for character, parts in pairs(originalCanCollide) do
        if character and character.Parent then
            for part, origValue in pairs(parts) do
                pcall(function()
                    if part and part.Parent then part.CanCollide = origValue end
                end)
            end
        end
    end
    originalCanCollide = {}
end

local function antiFlingGuard()
    if not antiFlingEnabled then return end
    if flingActive then return end
    if coinFarmWorking then return end
    if gunFarmWorking then return end
    if not canAct() then lastGoodSelf = nil return end
    local char, hrp = getChar()
    if not hrp then lastGoodSelf = nil return end
    if hrp.Position.X ~= hrp.Position.X then
        if lastGoodSelf then hrp.CFrame = lastGoodSelf end
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
        return
    end
    local lv = hrp.AssemblyLinearVelocity
    if lv.Magnitude > MAX_LINEAR then hrp.AssemblyLinearVelocity = lv.Unit * MAX_LINEAR end
    local av = hrp.AssemblyAngularVelocity
    if av.Magnitude > MAX_ANGULAR then hrp.AssemblyAngularVelocity = Vector3.zero end
    local now = tick()
    local exempt = isTeleporting() or (tick() - lastSelfTP < 0.35)
    if lastGoodSelf and not exempt then
        local dt = math.clamp(now - lastGoodSelfAt, 1/240, 0.5)
        local allow = math.max(ANCHOR_MIN, (MAX_LINEAR + 100) * dt)
        if (hrp.Position - lastGoodSelf.Position).Magnitude > allow then
            hrp.CFrame = lastGoodSelf
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
            return
        end
    end
    lastGoodSelf = hrp.CFrame
    lastGoodSelfAt = now
end

local function resetCharacterPhysics()
    pcall(function()
        local c = player.Character
        if not c then return end
        local hum = c:FindFirstChildOfClass("Humanoid")
        if hum then
            hum.PlatformStand = false
            hum.Sit = false
            hum:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
            pcall(function() hum:ChangeState(Enum.HumanoidStateType.GettingUp) end)
        end
        for _, p in ipairs(c:GetDescendants()) do
            if p:IsA("BasePart") then
                p.CanCollide = true
                p.Velocity = Vector3.zero
                p.RotVelocity = Vector3.zero
                p.AssemblyLinearVelocity = Vector3.zero
                p.AssemblyAngularVelocity = Vector3.zero
            end
        end
        if hum then
            workspace.CurrentCamera.CameraSubject = hum
        end
    end)
end

local function flingTarget(targetPlayer, duration)
    if not targetPlayer or not targetPlayer.Character then return false end
    if flingActive then return false end
    if not lockAcquire("fling") then return false end

    local targetChar = targetPlayer.Character
    local targetRoot = targetChar:FindFirstChild("HumanoidRootPart")
    local targetHum = targetChar:FindFirstChild("Humanoid")
    if not targetRoot or not targetHum or targetHum.Health <= 0 then
        lockRelease("fling")
        return false
    end

    local char, myRoot, myHum = getChar()
    if not char or not myRoot or not myHum then
        lockRelease("fling")
        return false
    end

    local flingStartChar = char
    local flingStartRoot = myRoot

    flingActive = true
    flingStartTime = tick()
    flingSafetySince = tick()
    duration = duration or 3

    local OldPos = myRoot.CFrame
    local oldCameraSubject = workspace.CurrentCamera.CameraSubject
    local savedFPDH = workspace.FallenPartsDestroyHeight

    pcall(function()
        if targetChar:FindFirstChild("Head") then
            workspace.CurrentCamera.CameraSubject = targetChar.Head
        else
            workspace.CurrentCamera.CameraSubject = targetHum
        end
    end)
    pcall(function() myHum:SetStateEnabled(Enum.HumanoidStateType.Seated, false) end)
    pcall(function() workspace.FallenPartsDestroyHeight = 0/0 end)

    local bv = Instance.new("BodyVelocity")
    bv.Name = "Flinger"
    bv.Parent = myRoot
    bv.Velocity = Vector3.new(900000000, 900000000, 900000000)
    bv.MaxForce = Vector3.new(1/0, 1/0, 1/0)

    local function applyCFrame(targetPart, cframeOffset, cframeRotation)
        if player.Character ~= flingStartChar then return end
        if not flingStartRoot.Parent then return end
        pcall(function()
            local baseCF = CFrame.new(targetPart.Position)
            flingStartRoot.CFrame = baseCF * cframeOffset * cframeRotation
            flingStartChar:SetPrimaryPartCFrame(baseCF * cframeOffset * cframeRotation)
            flingStartRoot.Velocity = Vector3.new(90000000, 900000000, 90000000)
            flingStartRoot.RotVelocity = Vector3.new(900000000, 900000000, 900000000)
        end)
    end

    local sum = 0
    local startTime = tick()

    pcall(function()
        while (tick() - startTime) < duration do
            task.wait()
            if player.Character ~= flingStartChar then break end
            if not flingStartRoot.Parent then break end
            if not targetRoot.Parent then break end
            if not targetHum.Parent or targetHum.Health <= 0 then break end

            pcall(function()
                if not (targetRoot.Velocity.Magnitude < 50) then
                    for _, item in ipairs({
                        {Vector3.new(0, 1.5, targetHum.WalkSpeed), math.rad(90)},
                        {Vector3.new(0, -1.5, -targetHum.WalkSpeed), 0},
                        {Vector3.new(0, 1.5, targetHum.WalkSpeed), math.rad(90)},
                        {Vector3.new(0, 1.5, targetRoot.Velocity.Magnitude / 1.25), math.rad(90)},
                        {Vector3.new(0, -1.5, -targetRoot.Velocity.Magnitude / 1.25), 0},
                        {Vector3.new(0, 1.5, targetRoot.Velocity.Magnitude / 1.25), math.rad(90)},
                        {Vector3.new(0, -1.5, 0), math.rad(90)},
                        {Vector3.new(0, -1.5, 0), 0},
                        {Vector3.new(0, -1.5, 0), math.rad(-90)},
                        {Vector3.new(0, -1.5, 0), 0},
                    }) do
                        applyCFrame(targetRoot, CFrame.new(item[1]), CFrame.Angles(item[2], 0, 0))
                        task.wait()
                    end
                else
                    sum = sum + 100
                    for _, item in ipairs({
                        Vector3.new(0, 1.5, 0),
                        Vector3.new(0, -1.5, 0),
                        Vector3.new(2.25, 1.5, -2.25),
                        Vector3.new(-2.25, -1.5, 2.25),
                        Vector3.new(0, 1.5, 0),
                        Vector3.new(0, -1.5, 0),
                    }) do
                        applyCFrame(
                            targetRoot,
                            CFrame.new(item) + targetHum.MoveDirection * (targetRoot.Velocity.Magnitude / 1.25),
                            CFrame.Angles(math.rad(sum), 0, 0)
                        )
                        task.wait()
                    end
                end

                if myHum and myHum.Parent then myHum.Sit = false end
                if targetChar:FindFirstChild("Head") then
                    workspace.CurrentCamera.CameraSubject = targetChar.Head
                end
            end)
        end
    end)

    pcall(function() if bv and bv.Parent then bv:Destroy() end end)
    pcall(function() if myHum and myHum.Parent then myHum:SetStateEnabled(Enum.HumanoidStateType.Seated, true) end end)
    pcall(function()
        workspace.CurrentCamera.CameraSubject = oldCameraSubject or (player.Character and player.Character:FindFirstChild("Humanoid"))
    end)

    local cleanupStart = tick()
    repeat
        if player.Character ~= flingStartChar then break end
        if not flingStartRoot or not flingStartRoot.Parent then break end

        pcall(function()
            flingStartRoot.CFrame = OldPos * CFrame.new(0, 0.5, 0)
            flingStartChar:SetPrimaryPartCFrame(OldPos * CFrame.new(0, 0.5, 0))
            local hum = flingStartChar:FindFirstChildOfClass("Humanoid")
            if hum then hum:ChangeState("GettingUp") end
            for _, p in ipairs(flingStartChar:GetChildren()) do
                if p:IsA("BasePart") then
                    p.Velocity = Vector3.zero
                    p.RotVelocity = Vector3.zero
                end
            end
        end)
        task.wait()
    until (tick() - cleanupStart > 3) 
       or (flingStartRoot.Position - OldPos.Position).Magnitude < 25

    pcall(function() workspace.FallenPartsDestroyHeight = savedFPDH end)

    pcall(function()
        local c = player.Character
        if c then
            local hum = c:FindFirstChildOfClass("Humanoid")
            if hum and hum.Sit then
                task.wait(0.5)
                hum.Sit = false
            end
        end
    end)

    flingActive = false
    flingStartTime = 0
    flingSafetySince = nil
    lockRelease("fling")
    return true
end

local function murdererAutoWin()
    local char, root, hum = getChar()
    if not char or not root or not hum then return end

    local bp = player:FindFirstChild("Backpack")
    local knife = (bp and bp:FindFirstChild("Knife")) or char:FindFirstChild("Knife")
    if not knife then
        warn("⚠️ مفيش سكين")
        return
    end
    if bp and bp:FindFirstChild("Knife") then
        hum:EquipTool(knife)
        task.wait(0.2)
    end

    local alive = {}
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= player and p.Character then
            local eR = p.Character:FindFirstChild("HumanoidRootPart")
            local eH = p.Character:FindFirstChild("Humanoid")
            if eR and eH and eH.Health > 0 and not deadPlayers[p.UserId] then
                alive[#alive + 1] = {plr = p, root = eR, hum = eH}
            end
        end
    end

    if #alive == 0 then return end

    local frontPos = (root.CFrame * CFrame.new(0, 0, -4)).Position

    for _, data in ipairs(alive) do
        pcall(function()
            data.root.CFrame = CFrame.new(frontPos)
            data.root.AssemblyLinearVelocity = Vector3.zero
            data.root.AssemblyAngularVelocity = Vector3.zero
        end)
    end
    task.wait(0.05)

    for _, data in ipairs(alive) do
        if data.hum.Health > 0 and data.root.Parent then
            for _ = 1, 4 do
                if data.hum.Health <= 0 then break end
                pcall(function()
                    root.CFrame = data.root.CFrame * CFrame.new(0, 0, 1.5)
                end)
                task.wait(0.02)
                pcall(function() knife:Activate() end)
                task.wait(0.02)
            end
        end
    end
end

                end
                local char, root, hum = getChar()
                if not char or not root or not hum then
                    task.wait(0.1)
                    return
                end

                local now = tick()

                if not platformStandDone then
                    local initialCoin = findNearestCoin(root)
                    if not initialCoin then
                        task.wait(0.2)
                        return
                    end
                    platformStandDone = true
                    lastSurfaceY = root.Position.Y
                    coinsWerePresent = true
                    pcall(function()
                        hum.PlatformStand = true
                        hum.WalkSpeed = COIN_SPEED
                    end)
                    pcall(function()
                        root.CFrame = CFrame.new(
                            initialCoin.Position.X,
                            initialCoin.Position.Y - COIN_DEPTH_UNDERGROUND,
                            initialCoin.Position.Z
                        )
                        root.AssemblyLinearVelocity = Vector3.zero
                        root.AssemblyAngularVelocity = Vector3.zero
                    end)
                    task.wait(0.1)
                    return
                end

                local stillValid = targetCoin and targetCoin.Parent
                    and not coinTaken(targetCoin)
                    and not (coinBlacklist[targetCoin] and now < coinBlacklist[targetCoin])

                if not stillValid or (now - lastScan > COIN_SCAN_INTERVAL) then
                    local newCoin = findNearestCoin(root)
                    if newCoin and newCoin ~= targetCoin then
                        targetCoin = newCoin
                        lastScan = now
                        lastCoinAcquire = now
                        coinsWerePresent = true
                    end
                end

                if not targetCoin or not targetCoin.Parent then
                    local remaining = freshCoins()
                    if #remaining > 0 then
                        coinsWerePresent = true
                    elseif #remaining == 0 and coinsWerePresent and not coinsAllCollected then
                        coinsAllCollected = true
                        task.spawn(runEndGameToggles)
                    end
                    task.wait(0.1)
                    return
                end

                if now - lastCoinAcquire > 5 then
                    coinBlacklist[targetCoin] = now + COIN_BLACKLIST_TIME
                    targetCoin = nil
                    lastScan = 0
                    task.wait(0.1)
                    return
                end

                local dt = RunService.RenderStepped:Wait()
                if not root.Parent then return end

                for _, part in ipairs(char:GetDescendants()) do
                    if part:IsA("BasePart") then part.CanCollide = false end
                end

                local coinPos = targetCoin.Position
                local underPos = Vector3.new(coinPos.X, coinPos.Y - COIN_DEPTH_UNDERGROUND, coinPos.Z)
                local myPos = root.Position

                local dx = underPos.X - myPos.X
                local dz = underPos.Z - myPos.Z
                local hDist = math.sqrt(dx*dx + dz*dz)

                if hDist > 1.5 then
                    local hStep = math.min(hDist, COIN_SPEED * dt)
                    local nx = myPos.X + (dx / hDist) * hStep
                    local nz = myPos.Z + (dz / hDist) * hStep
                    pcall(function()
                        root.CFrame = CFrame.new(nx, underPos.Y, nz)
                        root.AssemblyLinearVelocity = Vector3.zero
                        root.AssemblyAngularVelocity = Vector3.zero
                    end)
                    ThreadStatus.CoinFarm.lastBeat = tick()
                    return
                end

                local dxFull = coinPos.X - myPos.X
                local dyFull = coinPos.Y - myPos.Y
                local dzFull = coinPos.Z - myPos.Z
                local vDist = math.sqrt(dxFull*dxFull + dyFull*dyFull + dzFull*dzFull)

                if vDist > 1.5 then
                    local step = math.min(vDist, CLIMB_SPEED * dt)
                    local nx = myPos.X + (dxFull / vDist) * step
                    local ny = myPos.Y + (dyFull / vDist) * step
                    local nz = myPos.Z + (dzFull / vDist) * step
                    pcall(function()
                        root.CFrame = CFrame.new(nx, ny, nz)
                        root.AssemblyLinearVelocity = Vector3.zero
                        root.AssemblyAngularVelocity = Vector3.zero
                    end)

                    if firetouchinterest then
                        pcall(function() firetouchinterest(root, targetCoin, 0) end)
                        pcall(function() firetouchinterest(root, targetCoin, 1) end)
                    end
                    pcall(function()
                        local prompt = targetCoin:FindFirstChildWhichIsA("ProximityPrompt", true)
                        if prompt and fireproximityprompt then fireproximityprompt(prompt) end
                    end)
                    ThreadStatus.CoinFarm.lastBeat = tick()
                    return
                end

                if firetouchinterest then
                    for _ = 1, 5 do
                        pcall(function() firetouchinterest(root, targetCoin, 0) end)
                        pcall(function() firetouchinterest(root, targetCoin, 1) end)
                        local head = char:FindFirstChild("Head")
                        if head then
                            pcall(function() firetouchinterest(head, targetCoin, 0) end)
                            pcall(function() firetouchinterest(head, targetCoin, 1) end)
                        end
                    end
                end
                pcall(function()
                    local prompt = targetCoin:FindFirstChildWhichIsA("ProximityPrompt", true)
                    if prompt and fireproximityprompt then fireproximityprompt(prompt) end
                end)

                local backDist = (underPos - root.Position).Magnitude
                if backDist > 1.5 then
                    local stepDown = math.min(backDist, CLIMB_SPEED * dt)
                    local ddx = underPos.X - root.Position.X
                    local ddy = underPos.Y - root.Position.Y
                    local ddz = underPos.Z - root.Position.Z
                    pcall(function()
                        root.CFrame = CFrame.new(
                            root.Position.X + (ddx / backDist) * stepDown,
                            root.Position.Y + (ddy / backDist) * stepDown,
                            root.Position.Z + (ddz / backDist) * stepDown
                        )
                        root.AssemblyLinearVelocity = Vector3.zero
                        root.AssemblyAngularVelocity = Vector3.zero
                    end)
                    ThreadStatus.CoinFarm.lastBeat = tick()
                    return
                end

                if coinTaken(targetCoin) then
                    targetCoin = nil
                    lastScan = 0
                    local remaining = freshCoins()
                    if #remaining == 0 and coinsWerePresent and not coinsAllCollected then
                        coinsAllCollected = true
                        task.spawn(runEndGameToggles)
                    end
                end
                ThreadStatus.CoinFarm.lastBeat = tick()
            end)
            coinFarmWorking = false
            if not ok then
                ThreadStatus.CoinFarm.lastBeat = tick()
                task.wait(0.03)
            end
        end

        pcall(function()
            local _, root = getChar()
            if root and platformStandDone then
                local targetY = (lastSurfaceY or (root.Position.Y + 60)) + 5
                root.CFrame = CFrame.new(root.Position.X, targetY, root.Position.Z)
                root.AssemblyLinearVelocity = Vector3.zero
                root.AssemblyAngularVelocity = Vector3.zero
            end
        end)
        pcall(function()
            local _, _, hum = getChar()
            if hum then
                hum.PlatformStand = savedPS
                hum.WalkSpeed = savedSpeed
            end
        end)

        if Connections.CoinFarmNoclip then
            pcall(function() Connections.CoinFarmNoclip:Disconnect() end)
            Connections.CoinFarmNoclip = nil
        end
        if RunId.CoinFarm == myId then
            Running.CoinFarm = false
            ThreadStatus.CoinFarm.running = false
        end
    end)
end

local function enableGunFarm()
    RunId.GunFarm = RunId.GunFarm + 1
    Running.GunFarm = false
    gunFarmWorking = false
    Running.GunFarm = true
    RunId.GunFarm = RunId.GunFarm + 1
    local myId = RunId.GunFarm
    local myWatchdogId = WatchdogID
    ThreadStatus.GunFarm.running = true
    ThreadStatus.GunFarm.lastBeat = tick()
    task.spawn(function()
        while State.GunFarm and RunId.GunFarm == myId and getgenv().CCraftWatchdogID == myWatchdogId do
            ThreadStatus.GunFarm.lastBeat = tick()
            if flingActive then
                ThreadStatus.GunFarm.lastBeat = tick()
                task.wait(0.1)
                continue
            end
            gunFarmWorking = true
            local ok = pcall(function()
                if isSheriff(player) then return end
                if not canAct() then return end
                local char, root = getChar()
                if not char or not root then return end
                local drop, part = findGunDrop()
                if not drop or not part then return end
                local savedCF = root.CFrame
                local savedVel = root.Velocity
                local savedRotVel = root.RotVelocity
                markSelfTP()
                pcall(function()
                    root.CFrame = CFrame.new(part.Position)
                    root.Velocity = Vector3.new(0, 0, 0)
                end)
                pcall(function()
                    if firetouchinterest then
                        firetouchinterest(root, part, 0)
                        firetouchinterest(root, part, 1)
                    end
                end)
                pcall(function()
                    local prompt = drop:FindFirstChildWhichIsA("ProximityPrompt", true)
                    if prompt and fireproximityprompt then fireproximityprompt(prompt) end
                end)
                pcall(function()
                    if drop and drop.Parent and drop:IsA("Tool") then drop.Parent = player.Backpack end
                end)
                pcall(function()
                    local c, r = getChar()
                    if r then
                        r.CFrame = savedCF
                        r.Velocity = savedVel
                        r.RotVelocity = savedRotVel
                    end
                end)
            end)
            gunFarmWorking = false
            if not ok then
                ThreadStatus.GunFarm.lastBeat = tick()
                task.wait(0.02)
            end
            task.wait(0.02)
        end
        if RunId.GunFarm == myId then
            Running.GunFarm = false
            ThreadStatus.GunFarm.running = false
        end
    end)
end

    local myId = RunId.Aimbot
    local myWatchdogId = WatchdogID

    task.spawn(function()
        while State.Aimbot and RunId.Aimbot == myId and getgenv().CCraftWatchdogID == myWatchdogId do
            ThreadStatus.Aimbot.lastBeat = tick()
            pcall(function()
                if gunFarmWorking or coinFarmWorking then
                    ThreadStatus.Aimbot.lastBeat = tick()
                    task.wait()
                    return
                end
                
                if not gunReady() then
                    task.wait(0.1)
                    return
                end
                
                local murderer, eR, eH = findMurderer()
                if not murderer or not eR or not eH then 
                    task.wait(0.05) 
                    return 
                end
                
                local uid = murderer.UserId
                local prev = targetHistory[uid]
                
                local currentVel = eR.AssemblyLinearVelocity
                
                wallbangShoot(eR)
                
                if prev and gunReady() then
                    local velocitySpike = (currentVel - prev.velocity).Magnitude > 15
                    if velocitySpike then
                        task.wait(0.02)
                        if State.Aimbot and RunId.Aimbot == myId and gunReady() then
                            wallbangShoot(eR)
                        end
                    end
                end
                
                targetHistory[uid] = {velocity = currentVel}
            end)
            task.wait(0.05)
        end
        if RunId.Aimbot == myId then
            Running.Aimbot = false
            ThreadStatus.Aimbot.running = false
        end
    end)
end

local function addTrapESP(trapPart)
    if trapHighlights[trapPart] then return end
    if not trapPart:IsA("BasePart") then return end
    local hl = Instance.new("Highlight")
    hl.Name = "cCraftTrap"
    hl.Adornee = trapPart
    hl.FillColor = Color3.fromRGB(255, 90, 255)
    hl.FillTransparency = 0.4
    hl.OutlineColor = Color3.fromRGB(255, 170, 255)
    hl.OutlineTransparency = 0
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Enabled = false
    hl.Parent = trapPart
    trapHighlights[trapPart] = hl
end

local function removeTrapESP(trapPart)
    if trapHighlights[trapPart] then
        pcall(function() trapHighlights[trapPart]:Destroy() end)
        trapHighlights[trapPart] = nil
    end
end

task.spawn(function()
    for _, d in ipairs(workspace:GetDescendants()) do
        if d.Name == "TrapVisual" then pcall(addTrapESP, d) end
    end
    Connections.TrapAdded = workspace.DescendantAdded:Connect(function(d)
        if d.Name == "TrapVisual" then task.defer(addTrapESP, d) end
    end)
    Connections.TrapRemoved = workspace.DescendantRemoving:Connect(function(d)
        if trapHighlights[d] then removeTrapESP(d) end
    end)
end)

task.spawn(function()
    while true do
        task.wait(0.3)
        for trap, hl in pairs(trapHighlights) do
            if not trap.Parent then removeTrapESP(trap) else hl.Enabled = State.TrapESP end
        end
    end
end)

local function restoreAllState()
    forceResetAllRunners()
    task.wait(0.8)

    if State.Noclip then pcall(enableNoclip) end
    if State.AntiFling then pcall(enableAntiFling) end
    if anyESPMode() then pcall(enableESP) end
    if State.CoinFarm then pcall(enableCoinFarm) end
    if State.GunFarm then pcall(enableGunFarm) end
    if State.Aimbot then pcall(enableAimbot) end
    if State.ShootButton then pcall(function() if not shootGui then createShootButton() end end) end
    if State.ComboButton then pcall(function() if not comboGui then createComboButton() end end) end
end

local function mainWatchdog()
    LastWatchdogBeat = tick()
    pcall(function()
        if isShootPressed then
            if not isShootPressSince then isShootPressSince = tick()
            elseif (tick() - isShootPressSince > 15) then isShootPressed = false; isShootPressSince = nil end
        else isShootPressSince = nil end

        if comboRunning then
            if not comboSafetySince then comboSafetySince = tick()
            elseif (tick() - comboSafetySince > 8) then comboRunning = false; comboSafetySince = nil; lockRelease("combo") end
        else comboSafetySince = nil end

        if flingActive then
            if not flingSafetySince then flingSafetySince = tick()
            elseif (tick() - flingSafetySince > 10) then
                flingActive = false
                flingStartTime = 0
                flingSafetySince = nil
                lockRelease("fling")
                if flingAutoAntiFlingEnabled then pcall(disableAntiFling); flingAutoAntiFlingEnabled = false end
                resetCharacterPhysics()
            end
        else flingSafetySince = nil end

        if killAllRunning then
            if not killAllSafetySince then killAllSafetySince = tick()
            elseif (tick() - killAllSafetySince > 25) then killAllRunning = false; killAllSafetySince = nil; lockRelease("killall") end
        else killAllSafetySince = nil end

        if actionLock and not flingActive and not comboRunning and not killAllRunning and not gunFarmWorking and not coinFarmWorking then
            actionLock = nil
        end

        if not flingActive then
            local now = tick()
            if now - lastRoleSeenAt > 8 and now - lastRoleAutoRecoverAt > 8 then
                local m = findMurderer()
                local s = findSheriff()
                if m or s then
                    lastRoleSeenAt = now
                else
                    lastRoleAutoRecoverAt = now
                    clearRoleCaches()
                    lastRoleSeenAt = now
                end
            end
        end

        if State.CoinFarm then
            local needRestart = false
            if not ThreadStatus.CoinFarm.running then
                needRestart = true
            elseif (tick() - ThreadStatus.CoinFarm.lastBeat > 5) then
                if not flingActive then
                    needRestart = true
                else
                    ThreadStatus.CoinFarm.lastBeat = tick()
                end
            end
            if needRestart then
                RunId.CoinFarm = RunId.CoinFarm + 1
                Running.CoinFarm = false
                coinFarmWorking = false
                enableCoinFarm()
            end
        end

        if State.GunFarm then
            local needRestart = false
            if not ThreadStatus.GunFarm.running then
                needRestart = true
            elseif (tick() - ThreadStatus.GunFarm.lastBeat > 5) then
                if not flingActive then
                    needRestart = true
                else
                    ThreadStatus.GunFarm.lastBeat = tick()
                end
            end
            if needRestart then
                RunId.GunFarm = RunId.GunFarm + 1
                Running.GunFarm = false
                gunFarmWorking = false
                enableGunFarm()
            end
        end

        if State.Noclip and (not Connections.Noclip or not Connections.Noclip.Connected) then enableNoclip() end
        if State.AntiFling and not antiFlingEnabled then enableAntiFling() end
        if anyESPMode() then
            if (not ThreadStatus.ESP.running) or (tick() - ThreadStatus.ESP.lastBeat > 5) then
                RunId.ESP = RunId.ESP + 1
                Running.ESP = false
                enableESP()
            end
        end
        if State.Aimbot then
            if (not ThreadStatus.Aimbot.running) or (tick() - ThreadStatus.Aimbot.lastBeat > 5) then
                RunId.Aimbot = RunId.Aimbot + 1
                Running.Aimbot = false
                enableAimbot()
            end
        end
        
        -- ✅ ضمان: لو الحالة false والزر موجود → اقفله
        if not State.ShootButton and shootGui then
            removeShootButton()
        end
        if not State.ComboButton and comboGui then
            removeComboButton()
        end
    end)
end

local lastWatchdogCheck = 0
Connections.WatchdogHeartbeat = RunService.Heartbeat:Connect(function()
    if tick() - lastWatchdogCheck < 3 then return end
    lastWatchdogCheck = tick()
    task.spawn(mainWatchdog)
end)

task.spawn(function()
    local myWatchdogId = WatchdogID
    while getgenv().CCraftWatchdogID == myWatchdogId do
        task.wait(8)
        pcall(function()
            if tick() - LastWatchdogBeat > 12 then
                if not flingActive then
                    forceResetAllRunners()
                    if anyESPMode() then pcall(enableESP) end
                    if State.CoinFarm then pcall(enableCoinFarm) end
                    if State.GunFarm then pcall(enableGunFarm) end
                    if State.Aimbot then pcall(enableAimbot) end
                end
                LastWatchdogBeat = tick()
            end
        end)
    end
end)
